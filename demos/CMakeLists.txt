function(_awtk_c_demos_add_demo target)
  set(options "")
  set(oneValueArgs "")
  set(multiValueArgs SOURCES EXTRA_LIBS)
  cmake_parse_arguments(_demo "${options}" "${oneValueArgs}" "${multiValueArgs}" ${ARGN})

  add_executable(${target} ${_demo_SOURCES})
  awtk_c_demos_apply_compile_defs(${target})
  target_link_libraries(${target} PRIVATE awtk_c_demos_assets)
  if(_demo_EXTRA_LIBS)
    target_link_libraries(${target} PRIVATE ${_demo_EXTRA_LIBS})
  endif()
  set_target_properties(${target} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${_awtk_c_demos_runtime_dir}"
    LINKER_LANGUAGE CXX
  )
  if(MSVC)
    target_link_libraries(${target} PRIVATE legacy_stdio_definitions)
    set_target_properties(${target} PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
  if(UNIX)
    set_target_properties(${target} PROPERTIES
      INSTALL_RPATH "${_awtk_c_demos_rpath}"
      BUILD_WITH_INSTALL_RPATH FALSE
      MACOSX_RPATH ON
    )
    get_target_property(_awtk_imported awtk::awtk IMPORTED)
    if(_awtk_imported)
      get_target_property(_awtk_lib awtk::awtk IMPORTED_LOCATION)
      if(NOT _awtk_lib)
        get_target_property(_awtk_lib awtk::awtk IMPORTED_LOCATION_${CMAKE_BUILD_TYPE})
      endif()
      if(NOT _awtk_lib)
        get_target_property(_awtk_lib awtk::awtk IMPORTED_LOCATION_RELEASE)
      endif()
      if(_awtk_lib)
        get_filename_component(_awtk_lib_dir "${_awtk_lib}" DIRECTORY)
        set_target_properties(${target} PROPERTIES BUILD_RPATH "${_awtk_lib_dir}")
      endif()
    endif()
  endif()
endfunction()
